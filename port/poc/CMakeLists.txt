cmake_minimum_required(VERSION 3.20)
project(JFGRecompProof LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(JFG_PROOF_DIR "" CACHE PATH "Directory produced by prepare.py and N64Recomp")
if(NOT EXISTS "${JFG_PROOF_DIR}/poc_symbols.h")
    message(FATAL_ERROR "Run prepare.py and N64Recomp before configuring the proof")
endif()
file(GLOB JFG_GENERATED CONFIGURE_DEPENDS "${JFG_PROOF_DIR}/generated/*.c")
if(NOT JFG_GENERATED)
    message(FATAL_ERROR "No recompiled game functions found")
endif()
add_library(jfg_poc SHARED bridge.c ${JFG_GENERATED})
target_compile_definitions(jfg_poc PRIVATE JFG_POC_BUILD)
if(WIN32)
    set_target_properties(jfg_poc PROPERTIES PREFIX "")
endif()
target_include_directories(jfg_poc PRIVATE
    "${JFG_PROOF_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/N64ModernRuntime/N64Recomp/include")
if(MSVC)
    target_compile_options(jfg_poc PRIVATE /fp:strict)
else()
    target_compile_options(jfg_poc PRIVATE -fno-strict-aliasing -fwrapv -ffp-contract=off)
    target_link_libraries(jfg_poc PRIVATE m)
endif()
add_executable(jfg_poc_smoke smoke.c)
target_include_directories(jfg_poc_smoke PRIVATE "${JFG_PROOF_DIR}")
target_link_libraries(jfg_poc_smoke PRIVATE jfg_poc)
enable_testing()
add_test(NAME cpu_smoke COMMAND jfg_poc_smoke)
