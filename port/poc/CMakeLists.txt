cmake_minimum_required(VERSION 3.20)
project(JFGRecompProof LANGUAGES C)
option(JFG_BOOT_PROFILE "Include explicit headless ROM/cache platform bindings" OFF)
option(JFG_THREAD_PROFILE "Integrate cooperative runtime threads and message imports" OFF)
option(JFG_EVENT_PROFILE "Add session-owned clocks, timers and headless event routing" OFF)
option(JFG_INIT_PROFILE "Add PI DMA transport for original game initialization" OFF)
option(JFG_AUDIO_MANAGER_PROFILE "Add session-owned AI frequency configuration" OFF)
option(JFG_CONTROLLER_PROFILE "Add explicit headless controller-discovery test backend" OFF)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(JFG_PROOF_DIR "" CACHE PATH "Directory produced by prepare.py and N64Recomp")
if(NOT EXISTS "${JFG_PROOF_DIR}/poc_symbols.h")
    message(FATAL_ERROR "Run prepare.py and N64Recomp before configuring the proof")
endif()
file(GLOB JFG_GENERATED CONFIGURE_DEPENDS "${JFG_PROOF_DIR}/generated/*.c")
if(NOT JFG_GENERATED)
    message(FATAL_ERROR "No recompiled game functions found")
endif()
add_library(jfg_poc SHARED bridge.c ${JFG_GENERATED})
if(JFG_BOOT_PROFILE)
    target_sources(jfg_poc PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../boot/platform.c")
endif()
if(JFG_THREAD_PROFILE)
    if(NOT JFG_BOOT_PROFILE)
        message(FATAL_ERROR "The thread profile extends the boot profile")
    endif()
    enable_language(CXX)
    set_property(TARGET jfg_poc PROPERTY CXX_STANDARD 20)
    set_property(TARGET jfg_poc PROPERTY CXX_STANDARD_REQUIRED ON)
    find_package(Threads REQUIRED)
    set(JFG_RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/N64ModernRuntime")
    target_sources(jfg_poc PRIVATE
        ../threads/host.c ../threads/scheduler.cpp ../threads/kernel.cpp
        "${JFG_RUNTIME}/ultramodern/src/mesgqueue.cpp"
        "${JFG_RUNTIME}/ultramodern/src/scheduling.cpp")
    target_include_directories(jfg_poc PRIVATE ../threads
        "${JFG_RUNTIME}/ultramodern/include" "${JFG_RUNTIME}/thirdparty"
        "${JFG_RUNTIME}/thirdparty/concurrentqueue")
    target_link_libraries(jfg_poc PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(jfg_poc PRIVATE /EHa)
    else()
        # A runtime cancellation must unwind suspended generated C frames.
        target_compile_options(jfg_poc PRIVATE -fexceptions)
    endif()
    if(WIN32)
        target_compile_definitions(jfg_poc PRIVATE NOMINMAX)
        if(MINGW)
            target_link_options(jfg_poc PRIVATE -static-libstdc++ -static-libgcc)
            if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux")
                file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/compat")
                file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compat/Windows.h" "#include <windows.h>\n")
                target_include_directories(jfg_poc PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/compat")
            endif()
        endif()
    endif()
endif()
if(JFG_EVENT_PROFILE)
    if(NOT JFG_THREAD_PROFILE)
        message(FATAL_ERROR "The event profile requires the thread profile")
    endif()
    target_sources(jfg_poc PRIVATE ../events/host.c ../events/events.cpp)
    target_include_directories(jfg_poc PRIVATE ../events)
    target_compile_definitions(jfg_poc PRIVATE JFG_EVENT_PROFILE=1)
endif()
target_compile_definitions(jfg_poc PRIVATE JFG_POC_BUILD)
if(JFG_INIT_PROFILE)
    if(NOT JFG_EVENT_PROFILE)
        message(FATAL_ERROR "The initialization profile requires events and threads")
    endif()
    target_sources(jfg_poc PRIVATE ../init/host.c ../init/io.cpp)
    target_include_directories(jfg_poc PRIVATE ../init)
    target_compile_definitions(jfg_poc PRIVATE JFG_INIT_PROFILE=1)
endif()
if(JFG_AUDIO_MANAGER_PROFILE)
    if(NOT JFG_INIT_PROFILE)
        message(FATAL_ERROR "The audio manager profile requires initialization services")
    endif()
    target_sources(jfg_poc PRIVATE ../audio_manager/ai.c ../audio_manager/host.c)
    target_include_directories(jfg_poc PRIVATE ../audio_manager)
    target_compile_definitions(jfg_poc PRIVATE JFG_AUDIO_MANAGER_PROFILE=1)
endif()
if(JFG_CONTROLLER_PROFILE)
    if(NOT JFG_AUDIO_MANAGER_PROFILE)
        message(FATAL_ERROR "The controller bootstrap profile requires the audio manager services")
    endif()
    target_sources(jfg_poc PRIVATE ../controllers/controller.c ../controllers/host.c)
    target_include_directories(jfg_poc PRIVATE ../controllers)
    target_compile_definitions(jfg_poc PRIVATE JFG_CONTROLLER_PROFILE=1)
endif()
if(WIN32)
    set_target_properties(jfg_poc PROPERTIES PREFIX "")
endif()
target_include_directories(jfg_poc PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${JFG_PROOF_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/N64ModernRuntime/N64Recomp/include")
if(MSVC)
    target_compile_options(jfg_poc PRIVATE /fp:strict)
else()
    target_compile_options(jfg_poc PRIVATE -fno-strict-aliasing -fwrapv -ffp-contract=off)
    target_link_libraries(jfg_poc PRIVATE m)
endif()
add_executable(jfg_poc_smoke smoke.c)
target_include_directories(jfg_poc_smoke PRIVATE "${JFG_PROOF_DIR}")
target_link_libraries(jfg_poc_smoke PRIVATE jfg_poc)
enable_testing()
add_test(NAME cpu_smoke COMMAND jfg_poc_smoke)
