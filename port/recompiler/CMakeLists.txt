cmake_minimum_required(VERSION 3.20)
project(JFGRecompFrontend LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JFG_RECOMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/N64ModernRuntime/N64Recomp")
add_subdirectory("${JFG_RECOMP_ROOT}" n64recomp EXCLUDE_FROM_ALL)
add_executable(jfg_recomp_tool driver.cpp "${JFG_RECOMP_ROOT}/src/config.cpp")
target_include_directories(jfg_recomp_tool PRIVATE "${JFG_RECOMP_ROOT}/src")
target_link_libraries(jfg_recomp_tool PRIVATE N64Recomp N64RecompElf)
